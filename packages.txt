# sway (Wayland) Developer Setup - Complete Package List
# All packages required for this sway configuration

## Wayland / sway window manager
sway               # Wayland compositor, i3-compatible config
swaybg             # wallpaper
swayidle           # idle management (lock/dpms)
swaylock           # screen locker
waybar             # status bar (replaces polybar on Wayland)
fuzzel             # native Wayland app launcher (replaces rofi on Wayland)
grim               # screenshot capture
slurp              # region selector for screenshots
swappy             # screenshot annotation UI
wl-clipboard       # wl-copy / wl-paste (Wayland clipboard CLI)
cliphist           # clipboard history (replaces copyq)
xdg-desktop-portal-wlr  # wlroots portal backend (screenshare/screenshot)
wl-clip-persist    # optional: keeps clipboard after source app closes (CachyOS repo; AUR on stock Arch)

## Login (greetd — Wayland, no X server)
greetd             # minimal Wayland login daemon
greetd-regreet     # graphical greeter (wallpaper, remembers last user)
greetd-tuigreet    # console greeter (lightweight fallback)
cage               # tiny wlroots compositor that hosts regreet

## Session utilities (sway composites natively — no picom/compositor needed)
dunst              # notification daemon (Wayland-native)
dex                # XDG autostart

## Applications
kitty          # Terminal emulator (bound to Super+Return)
alacritty      # Terminal emulator (GPU, Wayland-native; font in config/alacritty)
zed            # Code editor
firefox        # Web browser
thunar         # File manager
thunar-archive-plugin
file-roller    # Archive manager
xarchiver      # Alternative archive manager
imv            # Image viewer (Wayland-native, replaces feh)
libwebp        # image loader: WEBP
libavif        # image loader: AVIF (common on modern sites)
libheif        # image loader: HEIC/HEIF
mupdf          # PDF viewer
mpv            # Media player
# (screenshots: grim + slurp + swappy — listed in the Wayland/sway section)

## Fonts & Typography
ttf-hack-nerd
ttf-firacode-nerd  # Kitty terminal font (with ligatures)
noto-fonts
noto-fonts-cjk     # Chinese, Japanese, Korean
noto-fonts-emoji   # Emoji support
noto-fonts-extra
ttf-liberation
ttf-dejavu

## Theming & Icons
papirus-icon-theme
materia-gtk-theme
breeze-gtk
gtk-engine-murrine

## System Utilities
brightnessctl      # Brightness control
pipewire           # Audio/video server (replaces PulseAudio)
pipewire-pulse     # PulseAudio compatibility layer (pactl, pavucontrol)
pipewire-alsa      # ALSA compatibility
wireplumber        # PipeWire session manager
pavucontrol       # Audio control GUI
networkmanager
network-manager-applet
xdg-utils
xdg-desktop-portal       # Portal (browser file dialogs, screenshare)
xdg-desktop-portal-gtk   # GTK backend for the portal
xdg-desktop-portal-wlr   # wlroots backend (screenshot/screencast on sway)
gvfs                     # Virtual FS for GTK file chooser (places/recent/mounts)
xorg-xwayland     # runs X11 apps inside sway (Steam, some Electron apps, etc.)
xorg-xrandr       # required by Steam; display query for XWayland apps
qt5ct             # Qt5 theme configuration
kvantum           # Qt theme engine

## Kernel (from CachyOS repo — REQUIRED for internal speakers on the ASUS TUF
## A14 FA401EA; stock Arch `linux` has no driver for its RT721 SDCA amp)
linux-cachyos
linux-cachyos-headers

## AMD GPU / Vulkan (Ryzen AI Max "Strix Halo", Radeon 8060S)
mesa
vulkan-radeon         # RADV open-source Vulkan driver for AMD
vulkan-icd-loader
vulkan-tools          # vulkaninfo / vkcube

## Gaming (requires [multilib] enabled in /etc/pacman.conf)
lib32-mesa
lib32-vulkan-radeon
steam

## Local AI / LLM (Vulkan path on the iGPU)
ollama-vulkan         # needs OLLAMA_IGPU_ENABLE=1 for Strix Halo (set via systemd drop-in)
llama-cpp-vulkan
